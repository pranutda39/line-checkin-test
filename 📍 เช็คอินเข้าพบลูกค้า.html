<body>

  <h2>📍 เช็คอินเข้าพบลูกค้า</h2>

  <p id="userName">กำลังโหลดข้อมูลผู้ใช้งาน...</p>

  

  <input type="text" id="locationName" placeholder="ระบุชื่อลูกค้า / ร้านค้า" />

  <p id="geoStatus" style="font-size: 12px; color: gray;"></p>

  

  <button id="btnCheckin" onclick="submitCheckin()" disabled>กำลังดึงพิกัด GPS...</button>


  <script>

    const GAS_URL = "https://script.google.com/macros/s/AKfycby8x6wv5GrCqSEZ9y7BdxOF30681EeOwnQqMcCdATnRJFZdN7Anx8MDGiJ2iz9zf8nA/exec"; // <-- วาง Web App URL จาก Google Apps Script ที่นี่

    let userProfile = null;

    let currentLat = null;

    let currentLng = null;


    async function main() {

      // 1. Initialize LIFF

      await liff.init({ liffId: "YOUR_LIFF_ID" }); // <-- ใส่ LIFF ID ในขั้นตอนที่ 3

      if (!liff.isLoggedIn()) {

        liff.login();

        return;

      }

      userProfile = await liff.getProfile();

      document.getElementById("userName").innerText = "สวัสดีคุณ: " + userProfile.displayName;


      // 2. Get GPS Location

      if (navigator.geolocation) {

        navigator.geolocation.getCurrentPosition(

          (pos) => {

            currentLat = pos.coords.latitude;

            currentLng = pos.coords.longitude;

            document.getElementById("geoStatus").innerText = `พิกัดปัจจุบัน: ${currentLat.toFixed(4)}, ${currentLng.toFixed(4)}`;

            document.getElementById("btnCheckin").disabled = false;

            document.getElementById("btnCheckin").innerText = "ยืนยัน Check-in";

          },

          (err) => {

            document.getElementById("geoStatus").innerText = "ไม่สามารถดึงตำแหน่ง GPS ได้ โปรดเปิด Location";

          },

          { enableHighAccuracy: true }

        );

      }

    }


    async function submitCheckin() {

      const locationName = document.getElementById("locationName").value;

      if (!locationName) {

        alert("กรุณากรอกชื่อลูกค้าหรือร้านค้าก่อนครับ");

        return;

      }


      document.getElementById("btnCheckin").disabled = true;

      document.getElementById("btnCheckin").innerText = "กำลังบันทึกข้อมูล...";


      const payload = {

        name: userProfile.displayName,

        locationName: locationName,

        lat: currentLat,

        lng: currentLng

      };


      await fetch(GAS_URL, {

        method: "POST",

        body: JSON.stringify(payload)

      });


      alert("เช็คอินสำเร็จ!");

      liff.closeWindow(); // ปิดหน้าต่าง LIFF กลับเข้าหน้าแชต LINE

    }


    main();

  </script>
